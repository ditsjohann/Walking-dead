# Phase 1 checks. Read the log: each `== label` says what the following lines should show.
# "Test failed" after a kill/ban check means the thing is gone, which is the expected result there.
== setup
gamerule spawn_mobs false
time set noon
player Bot spawn at 0 -57 0
wait 3
gamemode survival Bot
== starter kit (expect 1 axe, 1 pickaxe)
clear Bot minecraft:iron_axe 0
clear Bot minecraft:stone_pickaxe 0
== banned entities (creeper/skeleton/villager/xp fail, cow ok)
summon minecraft:creeper 10 -57 10
summon minecraft:skeleton 10 -57 10
summon minecraft:villager 10 -57 10
summon minecraft:experience_orb 10 -57 10
summon minecraft:cow 10 -57 10
wait 1
execute if entity @e[type=minecraft:creeper]
execute if entity @e[type=minecraft:villager]
execute if entity @e[type=minecraft:experience_orb]
execute if entity @e[type=minecraft:cow]
